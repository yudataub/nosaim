📖 חידוני מקור חיים לתלמידים

תצוגה מקדימה - קיצור שולחן ערוך מקור חיים לתלמידים (הרב חיים דוד הלוי זצ"ל)

🔑 מוּשָּׂגִים וּלְשׁוֹן חֲכָמִים – שַׁעַר רִאשׁוֹןפרקים א-ז: 13 מילים בתמונה, 19 מושגים, 20 מילים לכתיבה + 3 חזרה 🔑 מוּשָּׂגִים וּלְשׁוֹן חֲכָמִים – קַדִּישׁ, מִנְיָן, אָמֵן וּקְרִיאַת שְׁמַעשער שני, פרקים ח-יב 🔑 מוּשָּׂגִים וּלְשׁוֹן חֲכָמִים – שְׁמוֹנֶה עֶשְׂרֵה וְהַגְּשָׁמִיםשער שני, פרקים יג-טז 🔑 מוּשָּׂגִים וּלְשׁוֹן חֲכָמִים – חֲזָרַת הַשַּׁ״ץ, בִּרְכַּת כֹּהֲנִים, מִנְחָה וְעַרְבִיתשער שני, פרקים יז-כא 1. הַשְׁכָּמַת הַבֹּקֶרפרקים א-ב: קימה בבוקר, נטילת ידיים 2. אֲשֶׁר יָצַר וּבִרְכוֹת הַתּוֹרָהפרקים ג-ד: ברכת אשר יצר, ברכות התורה 3. בִּרְכוֹת הַשַּׁחַרפרק ה: סדר ברכות השחר 4. הַשְׁכָּמַת בֵּית הַכְּנֶסֶתפרק ו: קדושת בית הכנסת וההתנהגות בו 5. קַדִּישׁפרק ח (שער שני: תפילה): נוסח הקדיש וסוגיו 6. צִיצִית - חֵלֶק א׳פרק ז: טעם המצווה, זמנה, סדר העטיפה 7. הַטַּלִּית וְחוּטֵי הַצִּיצִיתפרק ז: מהבד ועד החוטים 8. טַלִּית שְׁאוּלָה וּפְתִיל תְּכֵלֶתפרק ז: חלק ג׳ (אחרון) 9. מִנְיַן עֲשָׂרָהפרק ט (שער שני): תפילה בציבור ודברים שבקדושה 10. נְטִילַת יָדַיִם לִפְנֵי הַסְּעוּדָהשער שלישי (ברכות), פרק כב 11. הַמּוֹצִיא וּבִרְכַּת הַמָּזוֹןפרקים כג-כד חלק א׳ 12. בִּרְכַּת הַמָּזוֹן - חֵלֶק ב׳פרק כד חלק ב׳ 13. בִּרְכַּת חֲמֵשֶׁת מִינֵי דָּגָןפרק כה 14. מֵעֵין שָׁלוֹשׁ וּפְרִי הָעֵץפרק כו חלק א׳ 15. שֶׁהַכֹּל, מְזוֹנוֹת וּמַשְׁקָאוֹתפרק כו חלק ב׳ 16. דִּין קְדִימָה בִּבְרָכוֹתפרק כז 17. מְכַוְּנִים אֶת הַלֵּב בַּבְּרָכָהפרק כח 18. בִּרְכַּת הָרֵיחַפרק כט 19. שֶׁהֶחֱיָנוּ וּבִרְכוֹת הַנִּסִּיםפרקים ל-לא 20. תְּפִלַּת הַדֶּרֶךְ וּזְהִירוּת בַּדְּרָכִיםפרק לב 21. בִּרְכַּת הַגּוֹמֵלפרק לג 22. בִּרְכוֹת הָרְאִיָּהפרקים לד-לה חלק א׳ 23. קֶשֶׁת, אִילָנוֹת וְחַמָּהפרק לה חלק ב׳ (סוף שער שלישי) 24. גְּדֻלַּת הַשַּׁבָּת וּכְבוֹדָהּשער רביעי (שבת), פרקים לו-לז 25. מְכִינִים אֶת הַשַּׁבָּתפרקים לח-לט 26. נֵרוֹת שַׁבָּתפרק מ 27. קַבָּלַת שַׁבָּתפרק מא 28. קִדּוּשׁ עַל הַיַּיִןפרק מב 29. שָׁלוֹשׁ סְעוּדוֹת וּתְפִלַּת מוּסָףפרקים מג-מד 30. קְרִיאַת הַתּוֹרָהפרק מה חלק א׳ 31. קְרִיאַת הַתּוֹרָה וְהַהַפְטָרָהפרק מה חלק ב׳ 32. עַרְבִית שֶׁל מוֹצָאֵי שַׁבָּתפרק מו 33. הַבְדָּלָה עַל הַכּוֹספרק מז חלק א׳ 34. הַבְדָּלָה וּמְלַוֶּה מַלְכָּהפרק מז חלק ב׳ 35. הַמְּלָאכוֹת הָאֲסוּרוֹת בְּשַׁבָּתפרק מח חלק א׳ 36. קוֹרֵעַ, בּוֹנֶה וְסוֹתֵרפרק מח חלק ב׳ 37. צָד, כּוֹתֵב וּמוֹחֵקפרק מח חלק ג׳ 38. בּוֹרֵר בְּשַׁבָּתפרק מט 39. בִּשּׁוּל בְּשַׁבָּתפרק נ 40. מִמְּצוֹא חֶפְצְךָ וְדַבֵּר דָּבָרפרק נא 41. אַרְבַּע רְשֻׁיּוֹת הַשַּׁבָּתפרק נב 42. מֻקְצֶה - חֵלֶק א׳פרק נג חלק א׳ 43. מֻקְצֶה - חֵלֶק ב׳פרק נג חלק ב׳ (סוף שער רביעי)
